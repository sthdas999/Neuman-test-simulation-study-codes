############################################################
# WEIGHTING FUNCTIONS
############################################################

weight_uniform <- function(x) {
  
  rep(
    1,
    length(x)
  )
}


weight_central <- function(x) {
  
  1 -
    (2 * x - 1)^2
}


weight_tail <- function(x) {
  
  (2 * x - 1)^2
}


############################################################
# LOCALIZED WEIGHT
############################################################

weight_localized <- function(
    x,
    x0,
    eta
) {
  
  exp(
    -(
      (x - x0)^2
    ) /
      (
        2 * eta^2
      )
  )
}


############################################################
# CENTERING CONSTANTS
############################################################

centering_constants <- list()


for (
  x0 in x0_values
) {
  
  for (
    eta in eta_values
  ) {
    
    grid <- seq(
      0,
      1,
      length.out = 10001
    )
    
    h_values <- exp(
      -(
        (grid - x0)^2
      ) /
        (
          2 * eta^2
        )
    )
    
    key <- paste(
      x0,
      eta,
      sep = "_"
    )
    
    centering_constants[[key]] <-
      mean(h_values)
  }
}


############################################################
# GENERATE LOCALIZED SAMPLE
############################################################

generate_localized_sample <- function(
    n,
    c,
    x0,
    eta,
    centering_constants = NULL
) {
  
  ##########################################################
  # KEY
  ##########################################################
  
  key <- paste(
    x0,
    eta,
    sep = "_"
  )
  
  
  ##########################################################
  # CENTERING CONSTANT
  ##########################################################
  
  if (
    !is.null(centering_constants) &&
    key %in% names(centering_constants)
  ) {
    
    center <-
      as.numeric(
        centering_constants[[key]]
      )
    
  } else {
    
    grid <- seq(
      0,
      1,
      length.out = 10001
    )
    
    h_values <- exp(
      -(
        (grid - x0)^2
      ) /
        (
          2 * eta^2
        )
    )
    
    center <-
      mean(h_values)
  }
  
  
  ##########################################################
  # CENTERED LOCALIZED PERTURBATION
  ##########################################################
  
  h_centered <- function(x) {
    
    exp(
      -(
        (x - x0)^2
      ) /
        (
          2 * eta^2
        )
    ) -
      center
  }
  
  
  ##########################################################
  # ALTERNATIVE DENSITY
  ##########################################################
  
  density_function <- function(x) {
    
    1 +
      (c / sqrt(n)) *
      h_centered(x)
  }
  
  
  ##########################################################
  # CHECK POSITIVITY
  ##########################################################
  
  grid_check <- seq(
    0,
    1,
    length.out = 10001
  )
  
  density_values <-
    density_function(
      grid_check
    )
  
  if (
    any(
      !is.finite(density_values)
    )
  ) {
    
    stop(
      "Non-finite density encountered."
    )
  }
  
  if (
    min(density_values) <= 0
  ) {
    
    stop(
      paste(
        "Non-positive density:",
        "n =", n,
        "c =", c,
        "x0 =", x0,
        "eta =", eta
      )
    )
  }
  
  
  ##########################################################
  # REJECTION SAMPLING
  ##########################################################
  
  maximum_density <-
    max(density_values)
  
  sample_x <- numeric(n)
  
  number_generated <- 0
  
  
  while (
    number_generated < n
  ) {
    
    remaining <-
      n -
      number_generated
    
    candidate <- runif(
      max(
        2 * remaining,
        10
      )
    )
    
    candidate_density <-
      density_function(
        candidate
      )
    
    acceptance_probability <-
      candidate_density /
      maximum_density
    
    accepted <-
      candidate[
        runif(
          length(candidate)
        ) <=
          acceptance_probability
      ]
    
    if (
      length(accepted) > 0
    ) {
      
      number_to_store <-
        min(
          length(accepted),
          remaining
        )
      
      index_start <-
        number_generated + 1
      
      index_end <-
        number_generated +
        number_to_store
      
      sample_x[
        index_start:index_end
      ] <-
        accepted[
          seq_len(
            number_to_store
          )
        ]
      
      number_generated <-
        number_generated +
        number_to_store
    }
  }
  
  
  return(
    sample_x
  )
}


############################################################
# SPACING STATISTIC
############################################################

spacing_statistic <- function(
    x,
    r,
    weight_function
) {
  
  x <- sort(x)
  
  n <- length(x)
  
  
  ##########################################################
  # VALIDITY CHECK
  ##########################################################
  
  if (
    r >= n
  ) {
    
    return(
      NA_real_
    )
  }
  
  
  ##########################################################
  # R-TH ORDER SPACINGS
  ##########################################################
  
  D_r <-
    x[
      (r + 1):n
    ] -
    x[
      1:(n - r)
    ]
  
  
  ##########################################################
  # MIDPOINTS
  ##########################################################
  
  midpoints <-
    (
      x[
        (r + 1):n
      ] +
      x[
        1:(n - r)
      ]
    ) / 2
  
  
  ##########################################################
  # WEIGHTS
  ##########################################################
  
  weights <-
    weight_function(
      midpoints
    )
  
  
  ##########################################################
  # CHECK WEIGHTS
  ##########################################################
  
  if (
    length(weights) !=
    length(D_r)
  ) {
    
    stop(
      "Weight function returned incorrect length."
    )
  }
  
  
  ##########################################################
  # CENTER SPACINGS
  ##########################################################
  
  Dbar_r <-
    mean(D_r)
  
  
  ##########################################################
  # NUMERATOR
  ##########################################################
  
  numerator <-
    sum(
      weights *
        (
          D_r -
            Dbar_r
        )^2
    )
  
  
  ##########################################################
  # DENOMINATOR
  ##########################################################
  
  denominator <-
    sum(
      (
        D_r -
          Dbar_r
      )^2
    )
  
  
  ##########################################################
  # CHECK DENOMINATOR
  ##########################################################
  
  if (
    !is.finite(denominator) ||
    denominator <= 0
  ) {
    
    return(
      NA_real_
    )
  }
  
  
  ##########################################################
  # STATISTIC
  ##########################################################
  
  statistic <-
    numerator /
    denominator
  
  
  return(
    statistic
  )
}


############################################################
# CRITICAL VALUE CALIBRATION
#
# ONLY THREE GLOBAL WEIGHTS:
# Uniform, Central, Tail
############################################################

critical_values <- numeric()


for (
  n in sample_sizes
) {
  
  for (
    r in r_values
  ) {
    
    ########################################################
    # ONLY r < n
    ########################################################
    
    if (
      r >= n
    ) {
      
      next
    }
    
    
    ########################################################
    # THREE GLOBAL WEIGHTS
    ########################################################
    
    weight_list <- list(
      Uniform = weight_uniform,
      Central = weight_central,
      Tail = weight_tail
    )
    
    
    ########################################################
    # CALIBRATE EACH GLOBAL WEIGHT
    ########################################################
    
    for (
      weight_name in names(weight_list)
    ) {
      
      null_values <-
        numeric(N)
      
      
      for (
        b in seq_len(N)
      ) {
        
        x_null <-
          runif(n)
        
        null_values[b] <-
          spacing_statistic(
            x = x_null,
            r = r,
            weight_function =
              weight_list[[weight_name]]
          )
      }
      
      
      null_values <-
        null_values[
          is.finite(null_values)
        ]
      
      
      if (
        length(null_values) == 0
      ) {
        
        stop(
          paste(
            "No valid null statistics:",
            "n =", n,
            "r =", r,
            "Weight =", weight_name
          )
        )
      }
      
      
      key <- paste(
        n,
        r,
        weight_name,
        sep = "_"
      )
      
      
      critical_values[key] <-
        as.numeric(
          quantile(
            null_values,
            probs = 1 - alpha,
            na.rm = TRUE
          )
        )
    }
    
    
    ########################################################
    # PROGRESS
    ########################################################
    
    cat(
      "Calibration completed:",
      "n =", n,
      "| r =", r,
      "\n"
    )
  }
}


############################################################
# POWER CALCULATION FUNCTION
############################################################

calculate_power <- function(
    n,
    r,
    weight_function,
    critical_value,
    generator
) {
  
  rejection <-
    logical(N)
  
  
  for (
    b in seq_len(N)
  ) {
    
    x <-
      generator()
    
    
    statistic <-
      spacing_statistic(
        x = x,
        r = r,
        weight_function =
          weight_function
      )
    
    
    rejection[b] <-
      is.finite(statistic) &&
      statistic >
      critical_value
  }
  
  
  power <-
    mean(rejection)
  
  
  mcse <-
    sqrt(
      power *
        (1 - power) /
        N
    )
  
  
  return(
    c(
      Power = power,
      MCSE = mcse
    )
  )
}
